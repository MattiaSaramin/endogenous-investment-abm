%------------------------------------------ numbered declaration environments
% The declaration boxes (formerly the ad-hoc `declbox`) are now numbered,
% \cref-able amsthm environments, still rendered in the same mdframed
% `declstyle` callout via mdframed's \newmdtheoremenv -- the box acquires a
% number, which is what makes it citable.  Two flavours: `modelassumption` for
% a declared structural choice, `remark` for a recorded observation or result.
% Defined AFTER cleveref, with explicit \crefname so \cref does not rely on
% auto-detection through \newmdtheoremenv.
\theoremstyle{definition}
\newmdtheoremenv[style=declstyle]{modelassumption}{Model assumption}
\theoremstyle{remark}
\newmdtheoremenv[style=declstyle]{remark}{Remark}
\crefname{modelassumption}{model assumption}{model assumptions}
\Crefname{modelassumption}{Model assumption}{Model assumptions}
\crefname{remark}{remark}{remarks}
\Crefname{remark}{Remark}{Remarks}
